cmake_minimum_required(VERSION 3.14)
project(best_viewer_native_tests LANGUAGES CXX)
get_filename_component(APP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(WEBP "${APP_ROOT}/tools/libwebp/windows/libwebp-1.4.0-windows-x64")
add_library(best_viewer_thumbnail_native SHARED "${APP_ROOT}/windows/runner/native_thumbnail.cpp")
target_compile_features(best_viewer_thumbnail_native PRIVATE cxx_std_17)
target_compile_definitions(best_viewer_thumbnail_native PRIVATE UNICODE _UNICODE NOMINMAX)
target_include_directories(best_viewer_thumbnail_native PRIVATE "${WEBP}/include")
target_link_libraries(best_viewer_thumbnail_native PRIVATE "${WEBP}/lib/libwebp.lib" windowscodecs ole32)
